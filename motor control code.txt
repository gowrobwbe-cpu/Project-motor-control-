#include <WebServer.h>

// =====================================================
// MYRAA - ESP32-S3 + L298N MOTOR CONTROL
// =====================================================

// ================= WIFI =================

const char* ssid = "MYRAA_ROBOT";
const char* password = "12345678";

WebServer server(80);


// =====================================================
// L298N PINS
// =====================================================

// RIGHT MOTOR
#define ENA 4
#define IN1 5
#define IN2 6

// LEFT MOTOR
#define ENB 7
#define IN3 15
#define IN4 16


// =====================================================
// SPEED
// =====================================================

int carSpeed = 255;

const int speedLow  = 150;
const int speedMid  = 200;
const int speedHigh = 255;


// =====================================================
// RIGHT MOTOR
// =====================================================

void rightForward() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  analogWrite(ENA, carSpeed);
}


void rightBackward() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  analogWrite(ENA, carSpeed);
}


void rightStop() {

  analogWrite(ENA, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
}


// =====================================================
// LEFT MOTOR
// =====================================================

void leftForward() {

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  analogWrite(ENB, carSpeed);
}


void leftBackward() {

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);

  analogWrite(ENB, carSpeed);
}


void leftStop() {

  analogWrite(ENB, 0);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


// =====================================================
// MOVEMENT
// =====================================================

// FORWARD
void forwardCar() {

  rightForward();
  leftForward();
}


// BACKWARD
void backwardCar() {

  rightBackward();
  leftBackward();
}


// LEFT
void leftCar() {

  rightForward();
  leftBackward();
}


// RIGHT
void rightCar() {

  rightBackward();
  leftForward();
}


// STOP
void stopCar() {

  rightStop();
  leftStop();
}

// =====================================================
// WEB HANDLERS
// =====================================================

void handleRoot() {

  server.send(
    200,
    "text/html",
    webpage()
  );
}


void handleForward() {

  forwardCar();

  server.send(
    200,
    "text/plain",
    "FORWARD"
  );
}


void handleBackward() {

  backwardCar();

  server.send(
    200,
    "text/plain",
    "BACKWARD"
  );
}


void handleLeft() {

  leftCar();

  server.send(
    200,
    "text/plain",
    "LEFT"
  );
}


void handleRight() {

  rightCar();

  server.send(
    200,
    "text/plain",
    "RIGHT"
  );
}


void handleStop() {

  stopCar();

  server.send(
    200,
    "text/plain",
    "STOP"
  );
}


// =====================================================
// SPEED
// =====================================================

void handleLow() {

  carSpeed = speedLow;

  server.send(
    200,
    "text/plain",
    "LOW"
  );
}


void handleMid() {

  carSpeed = speedMid;

  server.send(
    200,
    "text/plain",
    "MID"
  );
}


void handleHigh() {

  carSpeed = speedHigh;

  server.send(
    200,
    "text/plain",
    "HIGH"
  );
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);


  // Motor pins
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);


  // Stop at startup
  stopCar();


  // WiFi AP
  WiFi.mode(WIFI_AP);

  WiFi.softAP(
    ssid,
    password
  );


  delay(500);


  // Routes
  server.on(
    "/",
    handleRoot
  );

  server.on(
    "/F",
    handleForward
  );

  server.on(
    "/B",
    handleBackward
  );

  server.on(
    "/L",
    handleLeft
  );

  server.on(
    "/R",
    handleRight
  );

  server.on(
    "/S",
    handleStop
  );

  server.on(
    "/low",
    handleLow
  );

  server.on(
    "/mid",
    handleMid
  );

  server.on(
    "/high",
    handleHigh
  );


  server.begin();


  Serial.println();
  Serial.println("==============================");
  Serial.println("MYRAA ESP32-S3 MOTOR CONTROL");
  Serial.println("==============================");

  Serial.print("WiFi: ");
  Serial.println(ssid);

  Serial.print("IP: ");
  Serial.println(
    WiFi.softAPIP()
  );

  Serial.println(
    "Open: http://192.168.4.1"
  );
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  server.handleClient();

}
